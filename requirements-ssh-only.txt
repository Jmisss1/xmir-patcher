#

#   f981465ace35f96e0935b36091b61d17689d9e0e4f29b33a50882b92c45ddcbd

# install_ssh_only_venv.sh therefore builds OpenSSL 3.5.9 from
# https://github.com/openssl/openssl/releases/download/openssl-3.5.9/openssl-3.5.9.tar.gz
# sha256 603f5602e2eef00d77fbd429d34dcd5822bb301757a1bc9cdb24c670f1eb859a
# into .ssh-only-dist/openssl-prefix and points libssh2 only at that prefix.
# Linux confirmed that prefix's libcrypto.so.3 is what libssh2 linked. That was not macOS.
